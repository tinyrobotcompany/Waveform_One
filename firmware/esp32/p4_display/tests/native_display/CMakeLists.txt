cmake_minimum_required(VERSION 3.20)
project(waveform_native_display_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl-build)
file(GLOB FONT_SOURCES "../../main/fonts/waveform_font_*.c")
add_executable(native_display_check check.cpp ${FONT_SOURCES})
target_include_directories(native_display_check PRIVATE ../../main "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(native_display_check PRIVATE lvgl)
